cmake_minimum_required(VERSION 3.25)
project(uke-adreno-stats VERSION 1.0.15 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(ECM 6.0 REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs6)
find_package(Qt6 6.8 CONFIG REQUIRED COMPONENTS Core)
find_package(KF6CoreAddons 6.0 CONFIG REQUIRED)
find_package(KSysGuard 6.8 CONFIG REQUIRED)
add_library(uke_adreno MODULE plugin.cpp)
target_link_libraries(uke_adreno PRIVATE Qt6::Core KF6::CoreAddons KSysGuard::SystemStats)
target_compile_options(uke_adreno PRIVATE -Wall -Wextra -Werror)
install(TARGETS uke_adreno DESTINATION ${KDE_INSTALL_QTPLUGINDIR}/ksystemstats)
include(CTest)
if(BUILD_TESTING)
    add_executable(adreno-counter-test tests.cpp)
    target_compile_options(adreno-counter-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME adreno-counters COMMAND adreno-counter-test)
    add_executable(adreno-plugin-test plugin-test.cpp)
    target_compile_options(adreno-plugin-test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    target_link_libraries(adreno-plugin-test PRIVATE Qt6::Core KF6::CoreAddons KSysGuard::SystemStats)
    add_test(NAME adreno-plugin-load COMMAND adreno-plugin-test $<TARGET_FILE:uke_adreno>)
endif()
